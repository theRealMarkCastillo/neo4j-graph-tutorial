       ```python
       def _habits(self, now) -> list[dict]:
           MATCH (a:Entity {user:$user, kind:'activity'})<-[:ABOUT]-(m:Memory {user:$user})
           WHERE m.invalidated_at IS NULL AND m.consolidated_at IS NULL
             AND m.kind IN ['event','status','plan','feeling'] AND coalesce(m.status,'') <> 'lapsed'
             AND NOT coalesce(m.sensitive,false) AND NOT coalesce(m.private,false)
             AND m.created_at > $now - duration({days:$days})
             AND NOT coalesce(m.due_on > date($now), false)
           WITH a, m, coalesce(m.occurred_on, m.due_on, date(m.created_at)) AS day ORDER BY m.created_at
           WITH a, collect(m {.id,.text}) AS members, count(DISTINCT day) AS days WHERE days >= $min
           RETURN a.id AS activity, a.name AS name, members,
             COLLECT { MATCH (p:Memory {user:$user})-[:ABOUT]->(a) WHERE p.evidence IS NOT NULL AND p.invalidated_at IS NULL RETURN p.id ORDER BY p.created_at }[0] AS pattern
       ```
